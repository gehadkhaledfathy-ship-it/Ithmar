خطوة واحدة قبل التشغيل

  1. أنشئ مشروعًا مجانيًا في supabase.com.
  2. افتح SQL Editor، والصق الكود التالي، واضغط Run.
  3. من Project Settings ← API انسخ Project URL وanon key، وضعهما في أعلى السكربت في هذا الملف مكان PASTE، ثم ارفعه على GitHub.
create table fawaid (
  id bigint generated always as identity primary key,
  ch int not null check (ch between 2 and 12),
  body text not null check (char_length(body) <= 2000),
  lecture text, speaker text, author text,
  created_at timestamptz default now()
);
create table comments (
  id bigint generated always as identity primary key,
  fawaida_id bigint not null references fawaid(id) on delete cascade,
  body text not null check (char_length(body) <= 600),
  author text,
  created_at timestamptz default now()
);
create table admin (h text);
alter table fawaid enable row level security;
alter table comments enable row level security;
alter table admin enable row level security;
create policy "r1" on fawaid for select using (true);
create policy "w1" on fawaid for insert with check (true);
create policy "r2" on comments for select using (true);
create policy "w2" on comments for insert with check (true);

create function set_admin(code text) returns boolean
language plpgsql security definer set search_path = public as $$
begin
  if exists (select 1 from admin) or length(code) < 4 then return false; end if;
  insert into admin(h) values (encode(sha256(convert_to(code,'utf8')),'hex'));
  return true;
end $$;

create function delete_item(kind text, item_id bigint, code text) returns text
language plpgsql security definer set search_path = public as $$
declare cur text;
begin
  select h into cur from admin limit 1;
  if cur is null then return 'noadmin'; end if;
  if cur <> encode(sha256(convert_to(code,'utf8')),'hex') then return 'denied'; end if;
  if kind = 'f' then delete from fawaid where id = item_id;
  else delete from comments where id = item_id; end if;
  return 'ok';
end $$;
grant execute on function set_admin(text) to anon;
grant execute on function delete_item(text,bigint,text) to anon;
